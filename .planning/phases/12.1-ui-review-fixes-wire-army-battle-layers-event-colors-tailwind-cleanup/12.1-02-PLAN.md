```tsx
{/* Destruction Log */}
<div className="text-sm font-medium mb-2 mt-3" style={{ color: SIEGE_COLOR }}>
  Destruction Log
</div>
{(() => {
  // Find destroyed/ruined locations
  const destroyed = graph.getNodesByType('location')
    .filter(n => n.properties.isDestroyed || n.properties.destroyedTick != null)
    .sort((a, b) => ((b.properties.destroyedTick as number) ?? 0) - ((a.properties.destroyedTick as number) ?? 0))
    .slice(0, 10);
  if (destroyed.length === 0) {
    return <div className="text-[11px] opacity-60 italic">No destruction events yet.</div>;
  }
  return destroyed.map(loc => (
    <div key={loc.id} className="p-2 bg-[var(--bg-raised)] rounded mb-1.5 text-[11px]">
      <div className="font-semibold mb-1">{loc.name}</div>
      <div style={DETAIL_ROW_STYLE}>
        <span style={DETAIL_LABEL_STYLE}>Destroyed:</span>
        <span style={DETAIL_VALUE_STYLE}>Tick {loc.properties.destroyedTick ?? '?'}</span>
      </div>
      {loc.properties.destroyedBy && (
        <div style={DETAIL_ROW_STYLE}>
          <span style={DETAIL_LABEL_STYLE}>By:</span>
          <span style={DETAIL_VALUE_STYLE}>{String(loc.properties.destroyedBy)}</span>
        </div>
      )}
    </div>
  ));
})()}
```

NOTE: The exact property names for destruction data depend on what `phaseDestructionAftermath` (from Phase 12-06) writes to location nodes.
