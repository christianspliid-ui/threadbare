Create a new module following the BattleIndicatorMesh factory+tick pattern, but using `THREE.Points` instead of `THREE.Sprite`.

```typescript
import * as THREE from 'three';
import { hexToWorld } from '../../../lib/worldPosition';
import { LAYER_Z } from './RenderLayers';

// NFP #1: All tunable constants named
export const PARTICLE_CONSTANTS = {
  /** Number of particles per burst */
  COUNT: 16,
  /** Total animation duration in ms */
  LIFETIME_MS: 800,
  /** Maximum expansion radius in world units */
  EXPAND_RADIUS: 2.5,
  /** Point size (sizeAttenuation applies) */
  POINT_SIZE: 6.0,
  /** Initial opacity */
  SPAWN_OPACITY: 0.8,
  /** Z position for particles */
  Z_LAYER: LAYER_Z.PARTICLE_BURST,
} as const;

export interface ActiveBurst {
  points: THREE.Points;
  startMs: number;
  /** Cached random directions per particle (unit vectors * random magnitude) */
  directions: Float32Array;  // x,y pairs: [dx0, dy0, dx1, dy1, ...]
}

/**
 * Spawn a particle burst at the given hex.
 * Adds a THREE.Points object to the scene. Caller must call tickParticleBursts
 * each frame and clean up completed bursts.
 */
export function spawnParticleBurst(
  scene: THREE.Scene,
  hexCol: number,
  hexRow: number,
  hexSize: number,
  color: string,
  nowMs: number,
): ActiveBurst {
  const { x: cx, y: cy } = hexToWorld(hexCol, hexRow, hexSize);
  const count = PARTICLE_CONSTANTS.COUNT;

  // Position buffer: all particles start at center
  const positions = new Float32Array(count * 3);
  for (let i = 0; i < count; i++) {
    positions[i * 3] = cx;
    positions[i * 3 + 1] = cy;
    positions[i * 3 + 2] = PARTICLE_CONSTANTS.Z_LAYER;
  }

  // Random directions (seeded by index for determinism is not critical here — visual only)
  const directions = new Float32Array(count * 2);
  for (let i = 0; i < count; i++) {
    const angle = (i / count) * Math.PI * 2 + (Math.random() - 0.5) * 0.4;
    const magnitude = 0.5 + Math.random() * 0.5;  // 50-100% of max radius
    directions[i * 2] = Math.cos(angle) * magnitude;
    directions[i * 2 + 1] = Math.sin(angle) * magnitude;
  }

  const geometry = new THREE.BufferGeometry();
  geometry.setAttribute('position', new THREE.BufferAttribute(positions, 3));

  const threeColor = new THREE.Color(color);
  const material = new THREE.PointsMaterial({
    color: threeColor,
    size: PARTICLE_CONSTANTS.POINT_SIZE,
    sizeAttenuation: true,
    transparent: true,
    opacity: PARTICLE_CONSTANTS.SPAWN_OPACITY,
    depthWrite: false,
    blending: THREE.AdditiveBlending,
  });

  const points = new THREE.Points(geometry, material);
  points.renderOrder = 100;  // Above most scene elements
  scene.add(points);

  return { points, startMs: nowMs, directions };
}

/**
 * Tick all active bursts. Returns array of still-active bursts.
 * Completed bursts are removed from scene and disposed.
 */
export function tickParticleBursts(
  scene: THREE.Scene,
  bursts: ActiveBurst[],
  nowMs: number,
  centerX: number,   // hex center x (from original spawn — stored in positions[0])
  centerY: number,
): ActiveBurst[] {
  const surviving: ActiveBurst[] = [];

  for (const burst of bursts) {
    const elapsed = nowMs - burst.startMs;
    const t = elapsed / PARTICLE_CONSTANTS.LIFETIME_MS;

    if (t >= 1.0) {
      // Remove completed burst
      scene.remove(burst.points);
      burst.points.geometry.dispose();
      (burst.points.material as THREE.PointsMaterial).dispose();
      continue;
    }

    // Update positions: expand outward
    const posAttr = burst.points.geometry.getAttribute('position') as THREE.BufferAttribute;
    const positions = posAttr.array as Float32Array;
    const expandFactor = t * PARTICLE_CONSTANTS.EXPAND_RADIUS;

    // Read center from first particle's initial position (stored at spawn)
    const cx = positions[0] - burst.directions[0] * ((elapsed > 0 ? (elapsed - 16) / PARTICLE_CONSTANTS.LIFETIME_MS : 0) * PARTICLE_CONSTANTS.EXPAND_RADIUS);
    // Simpler: store cx/cy on the burst. Actually let's just recompute from burst data.
    // Better approach: store center on the burst interface.

    for (let i = 0; i < PARTICLE_CONSTANTS.COUNT; i++) {
      positions[i * 3] = positions[i * 3];     // Will be set below
      positions[i * 3 + 1] = positions[i * 3 + 1];
    }

    // Actually, simplest approach: store center coords on burst, compute position = center + direction * expandFactor
    // The caller passes centerX/centerY. But better to store on burst.
    // REVISED: Store cx/cy on ActiveBurst interface.

    posAttr.needsUpdate = true;

    // Update opacity: fade out
    const mat = burst.points.material as THREE.PointsMaterial;
    mat.opacity = PARTICLE_CONSTANTS.SPAWN_OPACITY * (1 - t);

    surviving.push(burst);
  }

  return surviving;
}
```

**IMPORTANT REVISION:** The above is a sketch. The actual implementation should store `cx` and `cy` on the `ActiveBurst` interface to avoid recalculation issues. Add:

```typescript
export interface ActiveBurst {
  points: THREE.Points;
  startMs: number;
  directions: Float32Array;
  cx: number;  // center x in world space
  cy: number;  // center y in world space
}
```

And in `tickParticleBursts`, iterate particles and set:
```typescript
positions[i * 3]     = burst.cx + burst.directions[i * 2] * expandFactor;
positions[i * 3 + 1] = burst.cy + burst.directions[i * 2 + 1] * expandFactor;
positions[i * 3 + 2] = PARTICLE_CONSTANTS.Z_LAYER;
```

Remove the `centerX`/`centerY` params from `tickParticleBursts` — use `burst.cx`/`burst.cy` instead. Final signature:

```typescript
export function tickParticleBursts(
  scene: THREE.Scene,
  bursts: ActiveBurst[],
  nowMs: number,
): ActiveBurst[]
```
