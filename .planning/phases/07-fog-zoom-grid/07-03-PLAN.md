    ### A. Create FollowMode.ts

    Create `src/components/HexMapV2/camera/FollowMode.ts`:

    ```typescript
    export interface FollowModeState {
      active: boolean;
      agentId: string | null;
      lastHexKey: string | null; // "col,row" of agent's last known hex
    }

    export function createFollowMode(): FollowModeState {
      return { active: false, agentId: null, lastHexKey: null };
    }

    export function updateFollowTarget(
      state: FollowModeState,
      agentId: string | null,
    ): void {
      state.active = agentId !== null;
      state.agentId = agentId;
      state.lastHexKey = null;
    }
    ```

    Simple mutable state object stored in a useRef. No React state — follow mode changes don't trigger re-renders.

    ### E. Replace scattered zoom threshold checks with visibility matrix

    In `zoom.on('zoom.labels', ...)` handler, replace the individual threshold checks with the visibility matrix:

    ```typescript
    import { getZoomTier, ZOOM_VISIBILITY_MATRIX, getFadeAlpha, FADE_RANGE, ZOOM_TIER_THRESHOLDS } from './scene/ZoomVisibilityMatrix';

    zoom.on('zoom.labels', (event) => {
      const k = event.transform.k;
      setZoomLevel(k);
      const tier = getZoomTier(k);

      // Layer visibility from matrix
      signifierGroup.visible = ZOOM_VISIBILITY_MATRIX.signifiers[tier];
      if (locationGroup) locationGroup.visible = ZOOM_VISIBILITY_MATRIX.locations[tier];
      elevTicksMesh.visible = ZOOM_VISIBILITY_MATRIX.elev_ticks[tier];
      if (riverGroup) riverGroup.visible = ZOOM_VISIBILITY_MATRIX.rivers[tier];
      if (roadGroup) roadGroup.visible = ZOOM_VISIBILITY_MATRIX.roads[tier];
      gridLines.visible = ZOOM_VISIBILITY_MATRIX.grid_lines[tier];

      // Agent tiers (portrait/dot/retinue) — existing updateZoomVisibility handles this
      updateZoomVisibility(agentSpriteGroup, k);

      // Trail visibility: same as agents_dot tier
      if (trailGroupRef.current) {
        trailGroupRef.current.visible = ZOOM_VISIBILITY_MATRIX.agents_dot[tier] || ZOOM_VISIBILITY_MATRIX.agents_portrait[tier];
      }

      // Border tiers
      if (borderKingdomMesh) borderKingdomMesh.visible = ZOOM_VISIBILITY_MATRIX.borders_kingdom[tier];
      if (borderBaronyMesh) borderBaronyMesh.visible = ZOOM_VISIBILITY_MATRIX.borders_barony[tier];

      // Fade transitions for sprite-based groups at tier boundaries
      // Apply opacity fade for signifiers/locations near the regional threshold (k=5)
      if (signifierGroup.visible) {
        const alpha = getFadeAlpha(k, ZOOM_TIER_THRESHOLDS.REGIONAL, FADE_RANGE * ZOOM_TIER_THRESHOLDS.REGIONAL);
        for (const child of signifierGroup.children) {
          if (child instanceof THREE.Sprite) {
            (child.material as THREE.SpriteMaterial).opacity = alpha;
          }
        }
      }
    });
    ```
