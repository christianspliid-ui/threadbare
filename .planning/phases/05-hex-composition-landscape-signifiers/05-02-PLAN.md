    **signifierTextures.ts** — SVG path data to Three.js texture:

    ```typescript
    import * as THREE from 'three';
    import type { SignifierVariant } from './signifierRegistry';

    // NFP #1: Tunable constants
    export const SIGNIFIER_TEXTURE_SIZE = 128; // px, rasterization resolution
    export const SIGNIFIER_FILL_COLOR = '#1a1a1a'; // pure black per locked decision

    export function buildSignifierTexture(variant: SignifierVariant, size: number = SIGNIFIER_TEXTURE_SIZE): THREE.CanvasTexture {
      const canvas = document.createElement('canvas');
      canvas.width = size;
      canvas.height = size;
      const ctx = canvas.getContext('2d')!;

      // Parse viewBox to get source dimensions (default 100x100)
      const [, , vbW, vbH] = variant.viewBox.split(' ').map(Number);
      const scaleX = size / (vbW || 100);
      const scaleY = size / (vbH || 100);

      ctx.save();
      ctx.scale(scaleX, scaleY);
      for (const path of variant.paths) {
        ctx.globalAlpha = path.opacity;
        ctx.fillStyle = SIGNIFIER_FILL_COLOR;
        ctx.fill(new Path2D(path.d));
      }
      ctx.restore();

      const tex = new THREE.CanvasTexture(canvas);
      tex.needsUpdate = true;
      return tex;
    }

    // Build all textures at startup — one per (terrainType, variantIndex)
    export function buildSignifierTextureCache(registry: SignifierRegistry): Map<string, THREE.CanvasTexture> {
      const cache = new Map<string, THREE.CanvasTexture>();
      for (const [terrain, variants] of Object.entries(registry)) {
        for (let i = 0; i < variants.length; i++) {
          const key = `${terrain}:${i}`;
          cache.set(key, buildSignifierTexture(variants[i]));
        }
      }
      return cache;
    }
    ```

    **SignifierMesh.ts** — Scene module following CoastlineMesh.ts pattern:

    ```typescript
    import * as THREE from 'three';
    import type { HexTile } from '../../../types';
    import { hexToPixel } from '../../../lib/hexMath';
    import { RENDER_ORDER } from './RenderLayers';
    import { HEX_CONSTANTS } from './HexFillMesh';
    import { SIGNIFIER_REGISTRY, TERRAIN_SIGNIFIER_FALLBACK, getSignifierParams } from '../signifiers/signifierRegistry';
    import { buildSignifierTextureCache } from '../signifiers/signifierTextures';

    // NFP #1: Tunable constants
    /** Sprite world-size at default scale (fraction of HEX_SIZE) */
    export const SIGNIFIER_SPRITE_SCALE = 0.7; // 70% of hex size
    /** Z offset for signifier sprites above borders */
    export const SIGNIFIER_Z = 0.07;

    // Water types that do NOT get signifiers
    const WATER_TYPES = new Set(['ocean','deep_ocean','tropical_ocean','coastal_shallows','coast','lake','river','reef']);

    export function createSignifierMesh(tiles: HexTile[], seed: number): THREE.Group {
      const group = new THREE.Group();
      group.renderOrder = RENDER_ORDER.SIGNIFIERS;

      // Build texture cache (one-time at scene init)
      const textureCache = buildSignifierTextureCache(SIGNIFIER_REGISTRY);

      const spriteSize = HEX_CONSTANTS.HEX_SIZE * SIGNIFIER_SPRITE_SCALE;

      for (const tile of tiles) {
        if (WATER_TYPES.has(tile.terrain)) continue;

        // Resolve terrain to registry key (direct or fallback)
        let registryKey = tile.terrain;
        if (!SIGNIFIER_REGISTRY[registryKey]) {
          registryKey = TERRAIN_SIGNIFIER_FALLBACK[registryKey] || '';
        }
        if (!registryKey || !SIGNIFIER_REGISTRY[registryKey]) continue; // fail-soft: no signifier for unknown type

        const variants = SIGNIFIER_REGISTRY[registryKey];
        const params = getSignifierParams(tile.col, tile.row, seed, variants.length);

        const texKey = `${registryKey}:${params.variantIndex}`;
        const texture = textureCache.get(texKey);
        if (!texture) continue; // fail-soft

        const material = new THREE.SpriteMaterial({
          map: texture,
          transparent: true,
          depthWrite: false,
        });

        const sprite = new THREE.Sprite(material);
        const { x, y } = hexToPixel(tile, HEX_CONSTANTS.HEX_SIZE);

        // Apply jitter (±10% of hex size)
        const jx = params.jitterX * HEX_CONSTANTS.HEX_SIZE;
        const jy = params.jitterY * HEX_CONSTANTS.HEX_SIZE;
        sprite.position.set(x + jx, -y + jy, SIGNIFIER_Z);

        // Apply rotation via material.rotation (radians)
        material.rotation = params.rotation;

        // Scale sprite
        sprite.scale.set(spriteSize, spriteSize, 1);

        group.add(sprite);
      }

      return group;
    }
    ```
