    ```typescript
    export type HexSlot = 'CENTER' | 'N' | 'NE' | 'SE' | 'S' | 'SW' | 'NW' | 'FILL' | 'RING';
    export type FootprintSize = 'full' | 'large' | 'medium' | 'small' | 'tiny';
    export type ZoomTier = 'hero-local' | 'regional' | 'continental' | 'full-world';

    export interface SuppressRule {
      target: string;  // entityType to suppress, e.g. 'terrain-signifier'
      when: 'always' | 'same-slot' | 'footprint-overlap';
    }

    export interface HexVisualManifest {
      entityType: string;
      preferredSlot: HexSlot;
      footprint: FootprintSize;
      suppresses: SuppressRule[];
      visibleAt: ZoomTier[];
      priority: number;  // higher = rendered first (gets preferred slot)
      fallbackSlots?: HexSlot[];
    }

    export interface CompositionResult {
      entity: string;  // entityType
      slot: HexSlot;
      visible: boolean;
    }
    ```

    **compositionResolver.ts** — Export `resolveHexComposition(entities: HexVisualManifest[]): CompositionResult[]`:
    1. Sort entities by priority descending
    2. Track occupied slots as Set<HexSlot>
    3. For each entity: try preferredSlot, then fallbackSlots in order, then mark hidden
    4. After slot assignment, evaluate suppression: for each entity with suppresses rules, check if any assigned entity matches target+when condition. If suppressed, set visible=false.
    5. Return CompositionResult[] in priority order

    NFP #2 (inspectability): Function is pure — same input always produces same output. No side effects.
    NFP #4 (fail-soft): Empty input returns []. Unknown entityTypes pass through without error.
