```typescript
/**
 * Quintessence — existential health meta-property.
 *
 * 0-1.0 continuous scale. All entities with SphereAffinity also carry quintessence.
 * NOT a reach. No axiological pair, no capability tier, no encounter affinity.
 * Displayed prose-only via IPK — never show numbers to player.
 */

import { QUINTESSENCE_LEXICON } from '../data/quintessence-content';

/** Event representing a change to an entity's quintessence */
export interface QuintessenceEvent {
  targetNodeId: string;
  delta: number;          // negative = erosion, positive = recovery
  source: string;         // e.g., 'overchannel', 'encounter_failure', 'passive_regen', 'doom'
  tick: number;
}

/** Threshold constants for quintessence state transitions */
export const QUINTESSENCE_THRESHOLDS = {
  WEAKENED: 0.25,
  CRITICAL: 0.10,
  DISSOLUTION: 0.0,
} as const;

/** Passive regeneration rate per tick (tunable — NFP #1) */
export const QUINTESSENCE_PASSIVE_REGEN = 0.002;

/** Default quintessence for entities missing the property (fail-soft) */
export const QUINTESSENCE_DEFAULT = 1.0;

/** Overchannel erosion amount per overchannel event (tunable) */
export const QUINTESSENCE_OVERCHANNEL_EROSION = 0.05;

/** Encounter failure erosion amount (tunable — NFP #1) */
export const QUINTESSENCE_ENCOUNTER_FAILURE_EROSION = 0.03;

/** Zero-state rules by entity category */
export const ZERO_STATE_RULES: Record<string, string> = {
  actor: 'death_or_transformation',
  location: 'ruins_or_void',
  faction: 'dissolution',
  default: 'removal',
} as const;

/**
 * Convert quintessence numeric value to prose word.
 * Uses QUINTESSENCE_LEXICON (10 levels, 0.0-1.0).
 */
export function quintessenceToWord(value: number): string {
  const clamped = Math.max(0, Math.min(1, value));
  const idx = Math.min(9, Math.floor(clamped * 10));
  return QUINTESSENCE_LEXICON[idx];
}
```

```typescript
/**
 * Quintessence tick phase — processes accumulated erosion/recovery events,
 * applies passive regeneration, and handles zero-state dissolution.
 *
 * Orchestrator position: after phaseSpherePressure (~9.7), before phaseSphereAggregation.
 */

import { GameState } from '../types/gameState';
import {
  QUINTESSENCE_PASSIVE_REGEN,
  QUINTESSENCE_DEFAULT,
  QUINTESSENCE_THRESHOLDS,
  ZERO_STATE_RULES,
} from '../types/quintessence';

export function phaseQuintessence(state: GameState): Partial<GameState> {
  const events = state.pendingQuintessenceEvents;
  const newTickEvents = [...state.tickEvents];
  const graph = state.graph; // mutate in place per existing convention

  // 1. Accumulate deltas per target
  const deltaMap = new Map<string, number>();
  for (const evt of events) {
    deltaMap.set(evt.targetNodeId, (deltaMap.get(evt.targetNodeId) ?? 0) + evt.delta);
  }

  // 2. Apply accumulated deltas
  for (const [nodeId, totalDelta] of deltaMap) {
    const node = graph.nodes.get(nodeId);
    if (!node) continue;
    const current = node.properties.quintessence ?? QUINTESSENCE_DEFAULT;
    node.properties.quintessence = Math.max(0, Math.min(1, current + totalDelta));
  }

  // 3. Passive regeneration for all entities with quintessence < 1.0
  for (const [, node] of graph.nodes) {
    if (node.properties.quintessence == null) continue; // only entities that have quintessence
    if (node.properties.quintessence < 1.0 && node.properties.quintessence > 0) {
      node.properties.quintessence = Math.min(1.0, node.properties.quintessence + QUINTESSENCE_PASSIVE_REGEN);
    }
  }

  // 4. Zero-state dissolution check
  for (const [nodeId, node] of graph.nodes) {
    if (node.properties.quintessence != null && node.properties.quintessence <= QUINTESSENCE_THRESHOLDS.DISSOLUTION) {
      const category = node.category ?? 'default';
      const rule = ZERO_STATE_RULES[category] ?? ZERO_STATE_RULES['default'];
      newTickEvents.push({
        type: 'dissolution_event',
        tick: state.tick,
        agentId: nodeId,
        details: `${node.properties.name ?? nodeId} has dissolved. Rule: ${rule}`,
      });
      // Mark as dissolved — do not remove from graph (fail-soft, let downstream handle)
      node.properties.dissolved = true;
    }
  }

  return {
    tickEvents: newTickEvents,
    pendingQuintessenceEvents: [],
  };
}
```
