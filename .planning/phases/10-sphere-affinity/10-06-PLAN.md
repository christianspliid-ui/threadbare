```tsx
interface ProseKeywordProps {
  sphere: SphereName;
  children: React.ReactNode;
}

export function ProseKeyword({ sphere, children }: ProseKeywordProps) {
  const [showTooltip, setShowTooltip] = useState(false);
  const tooltipId = useId();
  const color = getSphereColor(sphere); // from sphereColors.ts

  return (
    <span
      className="font-bold underline cursor-help inline"
      style={{ color }}
      onMouseEnter={() => setShowTooltip(true)}
      onMouseLeave={() => setShowTooltip(false)}
      onFocus={() => setShowTooltip(true)}
      onBlur={() => setShowTooltip(false)}
      role="term"
      tabIndex={0}
      aria-describedby={showTooltip ? tooltipId : undefined}
    >
      {children}
      {showTooltip && (
        <span
          id={tooltipId}
          role="tooltip"
          className="absolute z-50 px-3 py-2 text-sm bg-gray-900 text-gray-100 rounded-lg shadow-lg max-w-xs font-normal no-underline"
          style={{ /* position below the keyword */ }}
        >
          {SPHERE_TOOLTIPS[sphere]}
        </span>
      )}
    </span>
  );
}
```

```tsx
export function WorldSoulIndicator({ aggregate }: { aggregate?: SphereAggregate }) {
  if (!aggregate || !aggregate.dominantSphere) return null;

  const dominant = aggregate.dominantSphere;
  const avgScore = aggregate.totalBySphere[dominant] / Math.max(1, aggregate.entityCount);
  const intensity: Intensity =
    avgScore < 2 ? 'whisper' : avgScore < 4 ? 'murmur' : avgScore < 6 ? 'pulse' : 'storm';

  const proseTemplate = WORLD_SOUL_PROSE[dominant][intensity];
  return <div className="text-sm text-gray-300 italic">{renderProseWithIPK(proseTemplate)}</div>;
}
```

  ```tsx
  function actionSpherePreview(action: UnifiedActionTemplate, targetHexAffinity?: SphereAffinity): ReactNode {
    const sphere = action.sphereAffinity;
    if (!sphere || !targetHexAffinity) return null;

    const targetScore = targetHexAffinity.scores[sphere];
    const opposite = SPHERE_OPPOSITES[sphere];
    const oppositeScore = opposite ? (targetHexAffinity.scores[opposite] || 0) : 0;

    let prose: string;
    if (targetScore >= 5) {
      prose = `This land already resonates deeply with **${sphere}**. Your action will reinforce what is strong.`;
    } else if (oppositeScore >= 3) {
      prose = `The weight of **${opposite}** presses heavily here. Your **${sphere}** will face resistance.`;
    } else {
      prose = `The land is receptive. **${sphere.charAt(0).toUpperCase() + sphere.slice(1)}** will find purchase here.`;
    }

    return <p className="text-xs text-gray-400 italic mt-1">{renderProseWithIPK(prose)}</p>;
  }
  ```
