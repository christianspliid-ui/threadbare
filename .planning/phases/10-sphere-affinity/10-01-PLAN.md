```typescript
import { SphereAffinity, createDefaultSphereAffinity, getTerrainSphereScores, ARCHETYPE_SPHERE_BONUS_PRIMARY, ARCHETYPE_SPHERE_BONUS_SECONDARY, LOCATION_HEX_INHERIT_RATIO, LOCATION_TYPE_BONUS } from '../types/sphereAffinity';
import type { SphereName } from '../types/index';
import type { GraphNode } from '../types/graph';

/** Type-safe accessor for sphere affinity on any graph node */
export function getNodeSphereAffinity(node: GraphNode): SphereAffinity | undefined {
  return node.properties.sphereAffinity as SphereAffinity | undefined;
}

/** Seed hex sphere affinity from terrain type */
export function seedHexSphereAffinity(terrainType: string): SphereAffinity {
  const base = createDefaultSphereAffinity();
  const terrainScores = getTerrainSphereScores(terrainType);
  for (const [sphere, score] of Object.entries(terrainScores)) {
    base.scores[sphere as SphereName] = score;
  }
  return base;
}

/** Seed agent sphere affinity from sphereAlignment (existing CosmologyProfile) */
export function seedAgentSphereAffinity(sphereAlignment: Record<string, number> | undefined): SphereAffinity {
  const base = createDefaultSphereAffinity();
  if (!sphereAlignment) return base;
  // Find primary sphere (highest weight) and secondary (second highest)
  const sorted = Object.entries(sphereAlignment).sort((a, b) => b[1] - a[1]);
  if (sorted.length > 0 && sorted[0][1] > 0) {
    base.scores[sorted[0][0] as SphereName] = ARCHETYPE_SPHERE_BONUS_PRIMARY;
  }
  if (sorted.length > 1 && sorted[1][1] > 0) {
    base.scores[sorted[1][0] as SphereName] = ARCHETYPE_SPHERE_BONUS_SECONDARY;
  }
  return base;
}

/** Seed location from hex inheritance + type bias */
export function seedLocationSphereAffinity(
  hexAffinity: SphereAffinity,
  locationThematicSphere?: SphereName
): SphereAffinity {
  const base = createDefaultSphereAffinity();
  for (const sphere of SPHERE_NAMES) {
    base.scores[sphere] = Math.floor(hexAffinity.scores[sphere] * LOCATION_HEX_INHERIT_RATIO);
  }
  if (locationThematicSphere) {
    base.scores[locationThematicSphere] += LOCATION_TYPE_BONUS;
  }
  return base;
}
```
