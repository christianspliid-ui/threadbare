**Step 2: Create fantasy overlay pass** `src/engine/worldgen/passes/pass10-fantasyOverlay.ts`:
```typescript
/**
 * pass10-fantasyOverlay — Converts base biomes to magical variants based on sphere alignment.
 *
 * Applied after the pipeline completes, before toHexTilesFromContext.
 * Reads CosmologyProfile sphere weights and probabilistically upgrades terrain types.
 *
 * NFP #1: All thresholds are named constants.
 * NFP #3: Uses fractalNoise with PASS_SEED_FANTASY offset for determinism.
 * NFP #4: If cosmology is null/default, pass is a no-op.
 */

import type { CosmologyProfile } from '../../../types';
import type { TerrainType } from '../../../types';
import { fractalNoise } from '../../../lib/prng';
import { PASS_SEED_FANTASY } from '../constants';

export const FANTASY_OVERLAY_CONSTANTS = {
  /** Minimum sphere weight to trigger overlay effects */
  SPHERE_THRESHOLD: 0.65,
  /** Probability per eligible hex of being transformed */
  OVERLAY_CHANCE: 0.12,
  /** Noise scale for spatial coherence of overlay patches */
  NOISE_SCALE: 0.08,
};

/** Sphere-to-terrain transformation rules. Each rule: if sphere > threshold and terrain matches, transform. */
const OVERLAY_RULES: Array<{
  sphere: keyof CosmologyProfile;
  from: TerrainType[];
  to: TerrainType;
}> = [
  // High entropy → corruption/death
  { sphere: 'entropy', from: ['temperate_forest', 'dense_forest'], to: 'dead_forest' },
  { sphere: 'entropy', from: ['grassland', 'savanna'], to: 'broken_lands' },
  { sphere: 'entropy', from: ['hills', 'forested_hills'], to: 'badlands' },
  // High life → lush overgrowth
  { sphere: 'life', from: ['woodland', 'temperate_forest'], to: 'tropical_forest' },
  { sphere: 'life', from: ['grassland', 'steppe'], to: 'woodland' },
  { sphere: 'life', from: ['marsh'], to: 'swamp' },
  // High energy → volcanic activity
  { sphere: 'energy', from: ['rocky_desert', 'badlands'], to: 'volcanic' },
  { sphere: 'energy', from: ['mountains', 'high_mountains'], to: 'volcanic' },
  // High force → mountainous uplift
  { sphere: 'force', from: ['hills'], to: 'mountains' },
  { sphere: 'force', from: ['plateau'], to: 'high_mountains' },
  // High entropy + matter → frozen waste
  { sphere: 'entropy', from: ['tundra', 'snow_fields'], to: 'glacier' },
];

export function runFantasyOverlayPass(
  terrain: TerrainType[],
  cols: number,
  rows: number,
  seed: number,
  cosmology: CosmologyProfile | null | undefined,
): void {
  if (!cosmology) return; // NFP #4 fail-soft

  const { SPHERE_THRESHOLD, OVERLAY_CHANCE, NOISE_SCALE } = FANTASY_OVERLAY_CONSTANTS;

  for (let row = 0; row < rows; row++) {
    for (let col = 0; col < cols; col++) {
      const idx = row * cols + col;
      const baseTerrain = terrain[idx];

      // Deterministic noise per hex (NFP #3)
      const noise = (fractalNoise(
        col * NOISE_SCALE,
        row * NOISE_SCALE,
        seed + PASS_SEED_FANTASY,
        2, 0.5,
      ) + 1) / 2; // normalize to [0,1]

      if (noise >= OVERLAY_CHANCE) continue; // most hexes unaffected

      // Check each rule in priority order — first match wins
      for (const rule of OVERLAY_RULES) {
        const sphereWeight = cosmology[rule.sphere];
        if (sphereWeight > SPHERE_THRESHOLD && rule.from.includes(baseTerrain as TerrainType)) {
          terrain[idx] = rule.to;
          break;
        }
      }
    }
  }
}
```
