    import type { RegionLabel, RegionData, RegionCluster, BaronyRegion, KingdomRegion } from './regionTypes';
    import type { RiverPath } from './worldGenData';
    import { hexToPixel } from '../lib/hexMath';
    import { HEX_CONSTANTS } from '../components/HexMapV2/scene/HexFillMesh';
    import { REGION_MAP_LABEL_MIN_SIZE } from './regionDetection';

    // Label priority (lower = higher priority in collision detection)
    export const LABEL_PRIORITY: Record<RegionLabel['tier'], number> = {
      kingdom: 0,
      barony: 1,
      geographic: 2,
      river: 3,
    };

    export function generateRegionLabels(regionData: RegionData): RegionLabel[] {
      const labels: RegionLabel[] = [];

      // Kingdom labels
      for (const k of regionData.kingdoms) {
        const pos = hexToPixel(k.capitalHex, HEX_CONSTANTS.HEX_SIZE);
        labels.push({
          id: `kingdom-${k.id}`,
          tier: 'kingdom',
          text: k.name,
          worldX: pos.x,
          worldY: -pos.y, // Y-flip for Three.js
        });
      }

      // Barony labels (at centroid, not capital)
      for (const b of regionData.baronies) {
        const pos = hexToPixel({ col: b.centroid.col, row: b.centroid.row }, HEX_CONSTANTS.HEX_SIZE);
        labels.push({
          id: `barony-${b.id}`,
          tier: 'barony',
          text: b.name,
          worldX: pos.x,
          worldY: -pos.y,
        });
      }

      // Geographic labels (only for regions >= REGION_MAP_LABEL_MIN_SIZE)
      for (const g of regionData.geographicRegions) {
        if (g.hexes.length < REGION_MAP_LABEL_MIN_SIZE) continue;
        const pos = hexToPixel({ col: g.centerCol, row: g.centerRow }, HEX_CONSTANTS.HEX_SIZE);
        labels.push({
          id: `geo-${g.id}`,
          tier: 'geographic',
          text: g.name,
          worldX: pos.x,
          worldY: -pos.y,
        });
      }

      return labels;
    }

    ```typescript
    export interface ScreenLabel {
      id: string;
      tier: 'kingdom' | 'barony' | 'geographic' | 'river';
      text: string;
      screenX: number;
      screenY: number;
      visible: boolean;
    }

    export interface ScreenBBox {
      left: number; top: number; right: number; bottom: number;
    }

    // Font size estimates per tier for bbox computation
    const TIER_FONT_SIZE: Record<ScreenLabel['tier'], number> = {
      kingdom: 20,
      barony: 14,
      geographic: 12,
      river: 12,
    };

    export function estimateBBox(label: ScreenLabel): ScreenBBox {
      const fontSize = TIER_FONT_SIZE[label.tier];
      const charWidth = fontSize * 0.6; // approximate
      const width = label.text.length * charWidth;
      const height = fontSize * 1.4;
      return {
        left: label.screenX - width / 2,
        top: label.screenY - height / 2,
        right: label.screenX + width / 2,
        bottom: label.screenY + height / 2,
      };
    }

    function intersects(a: ScreenBBox, b: ScreenBBox): boolean {
      return a.left < b.right && a.right > b.left && a.top < b.bottom && a.bottom > b.top;
    }

    // Priority order: kingdom=0, barony=1, geographic=2, river=3
    const TIER_PRIORITY: Record<ScreenLabel['tier'], number> = {
      kingdom: 0, barony: 1, geographic: 2, river: 3,
    };

    export function removeOverlaps(labels: ScreenLabel[]): ScreenLabel[] {
      // Sort by priority (kingdom first)
      const sorted = [...labels].sort((a, b) => TIER_PRIORITY[a.tier] - TIER_PRIORITY[b.tier]);
      const placed: ScreenBBox[] = [];
      return sorted.map(label => {
        const bbox = estimateBBox(label);
        const overlaps = placed.some(p => intersects(bbox, p));
        if (!overlaps) {
          placed.push(bbox);
          return { ...label, visible: true };
        }
        return { ...label, visible: false };
      });
    }
    ```

    - **Zoom-tier filtering** (per CONTEXT.md locked decision):
      - Determine current zoom tier from zoomLevel. Thresholds (derive from CAMERA_CONSTANTS or define locally):
        - Hero-local: zoomLevel >= 15 (roughly 300px/hex)
        - Regional: zoomLevel >= 5 and < 15
        - Continental: zoomLevel >= 1.5 and < 5
        - Full-world: zoomLevel < 1.5
      - Kingdom labels: visible at continental + full-world (zoomLevel < 15)
      - Barony labels: visible at regional + continental (1.5 <= zoomLevel < 15)
      - Geographic labels: visible at regional only (5 <= zoomLevel < 15)
      - River labels: visible at regional only (5 <= zoomLevel < 15)
    - **Viewport culling:** Labels where sx < -100 or sx > canvasWidth+100 or sy < -100 or sy > canvasHeight+100 get `display: none`.
    - **Collision detection:** After projection + zoom filtering, call `removeOverlaps()` on visible labels. Hidden labels get `visibility: hidden` (not removed from DOM).
    - **Debounce:** Store projected positions in a ref. Only rerun collision detection when camera has moved significantly (debounce 60ms via a timestamp check, not setTimeout).
