```typescript
/** Named constants for merc seeding */
const MC_INSTANCE_COUNT = 2;
const MC_SEED_OFFSET = 51929; // distinct PRNG stream for merc companies
const MC_COMMANDER_IRON_CAP = 60; // Iron capability for placeholder commander
const MC_COMMANDER_GOLD_CAP = 40;

/**
 * Find the pair of location IDs with maximum hex distance.
 * Uses offset-to-axial conversion for cube distance.
 * O(n^2) but only runs once at seed time with ~10-30 qualifying locations.
 */
function findMaxDistancePair(
  graph: WorldGraph,
  ids: string[],
): [string, string] {
  if (ids.length < 2) return [ids[0], ids[0]];
  let bestA = ids[0], bestB = ids[1], bestDist = 0;
  for (let i = 0; i < ids.length; i++) {
    const a = graph.getNode(ids[i])?.properties;
    const colA = (a?.hexCol as number) ?? 0;
    const rowA = (a?.hexRow as number) ?? 0;
    // Offset (odd-q) to axial: q = col, r = row - (col - (col & 1)) / 2
    const qA = colA;
    const rA = rowA - (colA - (colA & 1)) / 2;
    for (let j = i + 1; j < ids.length; j++) {
      const b = graph.getNode(ids[j])?.properties;
      const colB = (b?.hexCol as number) ?? 0;
      const rowB = (b?.hexRow as number) ?? 0;
      const qB = colB;
      const rB = rowB - (colB - (colB & 1)) / 2;
      const dq = qA - qB;
      const dr = rA - rB;
      const ds = -dq - dr;
      const d = Math.max(Math.abs(dq), Math.abs(dr), Math.abs(ds));
      if (d > bestDist) { bestDist = d; bestA = ids[i]; bestB = ids[j]; }
    }
  }
  return [bestA, bestB];
}

/**
 * Seed two mercenary companies at maximum-distance settlements.
 * Each company gets: faction node, guild halls, static ambition, placeholder commander, army.
 */
function seedMercenaryCompanies(
  graph: WorldGraph,
  locationIds: readonly string[],
  seed: number,
  state: GameState,
): string[] {
  const def = MERCENARY_COMPANY_DEFINITION;

  // Find qualifying settlement locations (town/city/capital)
  const qualifying = locationIds.filter(id => {
    const node = graph.getNode(id);
    const sub = node?.properties.locationSubtype as string | undefined;
    return sub === 'town' || sub === 'city' || sub === 'capital';
  });

  if (qualifying.length < 2) return []; // fail-soft: not enough settlements

  const [locA, locB] = findMaxDistancePair(graph, qualifying);
  const primaryLocations = [locA, locB];
  const factionIds: string[] = [];

  for (let i = 0; i < MC_INSTANCE_COUNT; i++) {
    const instanceSuffix = String(i);
    const instanceSeed = seed + MC_SEED_OFFSET + i * 7919;

    // 1. Seed the faction node + guild halls (anchored to the primary location)
    const result = seedFactionFromDefinition(
      graph, def, locationIds, instanceSeed, instanceSuffix, primaryLocations[i],
    );
    factionIds.push(result.factionId);

    // 2. Rename the faction instance (differentiate the two companies)
    const companyNames = ['The Iron Wolves', 'The Scarlet Company'];
    graph.updateNode(result.factionId, { name: companyNames[i] });

    // 3. Seed static resource_acquisition ambition
    const ambitionId = `amb_${result.factionId}_seed`;
    graph.addNode({
      id: ambitionId,
      type: 'ambition',
      name: `${companyNames[i]} -- resource acquisition`,
      properties: {
        ambitionType: 'resource_acquisition',
        priority: 0.5,
        targetNodeId: null,
        grievanceDecay: 0,
        createdTick: 0,
      },
    });
    graph.addEdge({
      id: `e_pursues_${result.factionId}_seed`,
      source: result.factionId,
      target: ambitionId,
      type: 'pursues',
      properties: { priority: 0.5, status: 'active', milestones: [] },
    });

    // 4. Create placeholder commander agent (NPC belonging to the company)
    const commanderId = `agent_mc_cmdr_${i}`;
    graph.addNode({
      id: commanderId,
      type: 'actor',
      name: `${companyNames[i]} Commander`,
      properties: {
        actorType: 'individual',
        domainCapabilities: {
          iron: MC_COMMANDER_IRON_CAP,
          gold: MC_COMMANDER_GOLD_CAP,
          shadow: 20, veil: 10, heart: 15, eye: 15, stone: 20, star: 10,
        },
        displaced: false,
      },
    });
    // located_at: commander -> primary location
    graph.addEdge({
      id: `e_located_at_mc_cmdr_${i}`,
      source: commanderId,
      target: primaryLocations[i],
      type: 'located_at',
      properties: {},
    });
    // member_of: commander -> faction
    graph.addEdge({
      id: `e_member_of_mc_cmdr_${i}`,
      source: commanderId,
      target: result.factionId,
      type: 'member_of',
      properties: {
        role: 'commander',
        rank: 'war_chief',
        joinedTick: 0,
        factionDefId: def.id,
        reputation: 0.9,
      },
    });

    // 5. Spawn army at primary hall location
    spawnArmy(state, result.factionId, commanderId, ambitionId);
  }

  return factionIds;
}
```
