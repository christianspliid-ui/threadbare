**3. Add data extraction function** that reads army/battle/siege state from the game graph:
```typescript
function extractMilitaryRenderData(state: GameState): {
  armies: ArmyRenderData[];
  battles: BattleRenderData[];
  sieges: SiegeRenderData[];
} {
  const graph = state.graph;
  const armies: ArmyRenderData[] = [];
  const battles: BattleRenderData[] = [];
  const sieges: SiegeRenderData[] = [];

  // Extract armies from actor nodes with armyState
  const actorNodes = graph.getNodesByType('actor');
  for (const node of actorNodes) {
    if (node.properties.armyState != null) {
      const armyState = node.properties.armyState as import('../../types/army').ArmyState;
      // Get army position via located_at edge or direct properties
      // Follow the pattern used by existing agent position extraction
      const col = node.properties.col as number | undefined;
      const row = node.properties.row as number | undefined;
      if (col !== undefined && row !== undefined) {
        // Determine faction color from member_of edge
        const memberEdges = graph.getOutgoingEdges(node.id, 'member_of');
        const factionIndex = memberEdges[0] ? /* faction color index */ 0 : 0;
        armies.push({
          armyId: node.id,
          hexCol: col,
          hexRow: row,
          factionColor: FACTION_HERALDIC_COLORS[factionIndex % FACTION_HERALDIC_COLORS.length],
          armySize: armyState.size ?? 50,
          isInBattle: node.properties.battleState != null,
        });
      }
    }
    // Extract battles
    if (node.properties.battleState != null) {
      const bs = node.properties.battleState as BattleState;
      const col = node.properties.col as number | undefined;
      const row = node.properties.row as number | undefined;
      if (col !== undefined && row !== undefined) {
        battles.push({ battleNodeId: node.id, hexCol: col, hexRow: row });
        // Extract sieges from battle state with settlementId
        if (bs.settlementId) {
          const settlementNode = graph.getNode(bs.settlementId);
          const sCol = settlementNode?.properties.col as number | undefined;
          const sRow = settlementNode?.properties.row as number | undefined;
          if (sCol !== undefined && sRow !== undefined) {
            const attackerEdges = graph.getOutgoingEdges(bs.attackerArmyId, 'member_of');
            sieges.push({
              siegeNodeId: node.id,
              settlementHexCol: sCol,
              settlementHexRow: sRow,
              factionColor: FACTION_HERALDIC_COLORS[0], // derive from attacker faction
            });
          }
        }
      }
    }
  }
  return { armies, battles, sieges };
}
```

Adjust the faction color derivation to match how the existing agent adapter maps faction indices to FACTION_HERALDIC_COLORS. Check `src/components/HexMapV2/` for the existing adapter pattern.
